function setServerIcon(url) { const mainIcon = document.getElementById("mainServerIcon"); const sidebarIcon = document.getElementById("sidebarServerIcon"); const fallback = () => { if (mainIcon) { mainIcon.style.display = "none"; } if (sidebarIcon) { sidebarIcon.style.display = "none"; } }; if (!url) { fallback(); return; } [mainIcon, sidebarIcon].forEach((img) => { if (!img) return; img.style.display = "block"; img.src = url; img.onerror = () => { img.style.display = "none"; }; }); } function getServerIconUrl(server) { if (!server || !server.id || !server.icon) { return null; } const iconHash = String(server.icon); if (iconHash.startsWith("http")) { return iconHash; } const extension = iconHash.startsWith("a_") ? "gif" : "png"; return `https://cdn.discordapp.com/icons/${server.id}/${iconHash}.${extension}?size=256`; } // Después de obtener los datos del servidor: const iconUrl = getServerIconUrl(server); setServerIcon(iconUrl);
